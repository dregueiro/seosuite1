const bsModal = new bootstrap.Modal(document.getElementById('modalCredential'));

// GESTIÓN DE CREDENCIALES
function openCredModal(mode) {
    const credSelect = document.getElementById('id_google_credential');
    const credId = credSelect.value;

    if (mode === 'create') {
        document.getElementById('modalTitle').innerText = "Nueva Service Account";
        document.getElementById('cred_id').value = "";
        document.getElementById('cred_name').value = "";
        document.getElementById('cred_json').value = "";
        bsModal.show();
    } else {
        if (!credId) { alert("Selecciona una credencial primero."); return; }
        fetch(`/integrations/credentials/${credId}/get/`)
            .then(res => res.json())
            .then(data => {
                document.getElementById('modalTitle').innerText = "Editar Credencial: " + data.name;
                document.getElementById('cred_id').value = data.id;
                document.getElementById('cred_name').value = data.name;
                document.getElementById('cred_json').value = JSON.stringify(data.json, null, 2);
                bsModal.show();
            });
    }
}

function saveCredential() {
    const id = document.getElementById('cred_id').value;
    const payload = {
        name: document.getElementById('cred_name').value,
        service_account_json: document.getElementById('cred_json').value
    };
    if (!payload.name || !payload.service_account_json) return alert("Completa los campos.");

    const url = id ? `/integrations/credentials/${id}/update/` : "/integrations/credentials/quick-create/";

    fetch(url, {
        method: "POST",
        headers: { "Content-Type": "application/json", "X-CSRFToken": "{{ csrf_token }}" },
        body: JSON.stringify(payload)
    }).then(res => res.json()).then(data => {
        if (data.status === 'success') location.reload();
        else alert("Error: " + data.message);
    });
}

function deleteCredential() {
    const credId = document.getElementById('id_google_credential').value;
    if (!credId) return alert("Selecciona una credencial.");
    if (confirm("¿Eliminar permanentemente?")) {
        fetch(`/integrations/credentials/${credId}/delete/`, {
            method: "POST",
            headers: { "X-CSRFToken": "{{ csrf_token }}" }
        }).then(() => location.reload());
    }
}

const bsModal = new bootstrap.Modal(document.getElementById('modalCredential'));

// GESTIÓN DE CREDENCIALES
function openCredModal(mode) {
    const credSelect = document.getElementById('id_google_credential');
    const credId = credSelect.value;

    if (mode === 'create') {
        document.getElementById('modalTitle').innerText = "Nueva Service Account";
        document.getElementById('cred_id').value = "";
        document.getElementById('cred_name').value = "";
        document.getElementById('cred_json').value = "";
        bsModal.show();
    } else {
        if (!credId) { alert("Selecciona una credencial primero."); return; }
        fetch(`/integrations/credentials/${credId}/get/`)
            .then(res => res.json())
            .then(data => {
                document.getElementById('modalTitle').innerText = "Editar Credencial: " + data.name;
                document.getElementById('cred_id').value = data.id;
                document.getElementById('cred_name').value = data.name;
                document.getElementById('cred_json').value = JSON.stringify(data.json, null, 2);
                bsModal.show();
            });
    }
}

function saveCredential() {
    const id = document.getElementById('cred_id').value;
    const payload = {
        name: document.getElementById('cred_name').value,
        service_account_json: document.getElementById('cred_json').value
    };
    if (!payload.name || !payload.service_account_json) return alert("Completa los campos.");

    const url = id ? `/integrations/credentials/${id}/update/` : "/integrations/credentials/quick-create/";

    fetch(url, {
        method: "POST",
        headers: { "Content-Type": "application/json", "X-CSRFToken": "{{ csrf_token }}" },
        body: JSON.stringify(payload)
    }).then(res => res.json()).then(data => {
        if (data.status === 'success') location.reload();
        else alert("Error: " + data.message);
    });
}

function deleteCredential() {
    const credId = document.getElementById('id_google_credential').value;
    if (!credId) return alert("Selecciona una credencial.");
    if (confirm("¿Eliminar permanentemente?")) {
        fetch(`/integrations/credentials/${credId}/delete/`, {
            method: "POST",
            headers: { "X-CSRFToken": "{{ csrf_token }}" }
        }).then(() => location.reload());
    }
}
